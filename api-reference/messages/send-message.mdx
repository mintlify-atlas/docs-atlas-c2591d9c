---
title: "Send message"
description: "Post a new message to a channel. Optionally trigger an AI response by including @ai in the message content."
---

```bash
POST /chat/channel/{uuid}/message
```

<Info>Authentication is required. You must be a member of the channel to send messages.</Info>

## Path parameters

<ParamField path="uuid" type="string" required>
  The UUID of the channel to send the message to.
</ParamField>

## Body parameters

<ParamField body="content" type="string" required>
  The text content of the message. Must be at least 1 character.
</ParamField>

<Note>
  If `content` includes `@ai` and the channel has AI enabled, an AI response is automatically generated and published to the channel after your message is saved.
</Note>

## Response

Returns the created message object.

<ResponseField name="uuid" type="string">
  Unique identifier of the message.
</ResponseField>

<ResponseField name="senderId" type="string">
  ID of the user who sent the message.
</ResponseField>

<ResponseField name="content" type="string">
  Text content of the message.
</ResponseField>

<ResponseField name="channelUuid" type="string">
  UUID of the channel this message belongs to.
</ResponseField>

<ResponseField name="createdAt" type="string">
  ISO 8601 timestamp of when the message was created.
</ResponseField>

## Errors

| Status | Code | Description |
| --- | --- | --- |
| 401 | `UNAUTHORIZED` | The request is not authenticated. |
| 403 | `FORBIDDEN` | You are not a member of this channel. |
| 500 | `INTERNAL_SERVER_ERROR` | Something went wrong. |

## TypeScript example

```typescript
import { orpc } from '@/lib/orpc'
import { useMutation } from '@tanstack/react-query'

function SendMessageForm({ channelUuid }: { channelUuid: string }) {
  const { mutate: sendMessage } = useMutation(
    orpc.chat.message.sendMessageToChannel.mutationOptions()
  )

  function handleSubmit(content: string) {
    sendMessage({ uuid: channelUuid, content })
  }

  return (
    <button onClick={() => handleSubmit('Hello, world!')}>
      Send message
    </button>
  )
}
```
